  helm upgrade {{ .Release.Name }} ./k8s/moviemate \

    -n {{ .Release.Namespace }}

  # Desinstalar (los PVCs de PostgreSQL se conservan)
  helm uninstall {{ .Release.Name }} -n {{ .Release.Namespace }}
